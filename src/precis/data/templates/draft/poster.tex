% Generic poster scaffold. Content slots are deliberately empty; no assets required.
% A0 portrait: 84.1 x 118.9 cm. Font scale is 1.13; use sans-serif Latin Modern.
% Build standalone: tectonic -X compile poster.tex
% Alternatively: latexmk -lualatex -interaction=nonstopmode -halt-on-error poster.tex
% Product draft compilation continues to use its existing LuaLaTeX sandbox.
% Larger print variant: replace the beamerposter options below with
%   size=custom,orientation=portrait,width=92,height=130,scale=1.13
% Width/height are centimetres; size=custom is REQUIRED or A0 wins silently.
% Verify the output: pdfinfo poster.pdf | grep 'Page size'
% Fit check: tectonic -X compile poster.tex 2>&1 | grep 'Overfull .vbox'
% A successful compile alone does not prove fit: inspect the log and rendered page.
% Re-check after changing size, font, header/footer or content. colheight is a
% measured vertical budget, not an automatic page calculation; tune it to fit.
% Minipage alignment [s] distributes vfill BETWEEN blocks; [t] does not.
% The 0.95 inset and 0.330/0.315/0.330 columns preserve the supplied geometry.
% Branding is separate data: these neutral roles are product defaults only.
% No institution palettes, logo paths, funder marks or private files belong here.
\documentclass[final]{beamer}
\usepackage[size=a0,orientation=portrait,scale=1.13]{beamerposter}
\usepackage{lmodern}
\usepackage[T1]{fontenc}
\usepackage{amsmath,amssymb}
\usepackage{graphicx}
\usepackage{booktabs,array}
\usepackage{upgreek}
\usepackage{multicol}
\hypersetup{hidelinks,pdfborder={0 0 0},pdftitle={},pdfauthor={}}
\usetheme{default}
\usecolortheme{default}
\setbeamertemplate{navigation symbols}{}
\setbeamertemplate{headline}{}
\setbeamertemplate{footline}{}

% Declared colour roles — the neutral product defaults. A validated
% branding profile redefines each role by injecting
% \definecolor{<role>}{HTML}{<hex>} ahead of the document body; every rule
% below names a role, never a hue, so that is the ONLY thing a rebrand
% touches.
\definecolor{accent}{HTML}{404040}
\definecolor{heading}{HTML}{262626}
\definecolor{band}{HTML}{262626}
\definecolor{bodytint}{HTML}{F5F5F5}
\definecolor{citation}{HTML}{4D4D4D}
\definecolor{measured}{HTML}{333333}
\definecolor{computed}{HTML}{595959}
\definecolor{bandtext}{HTML}{FFFFFF}
\definecolor{bodytext}{HTML}{000000}
\setbeamercolor{normal text}{fg=bodytext,bg=white}
\setbeamercolor{block title}{fg=bandtext,bg=heading}
\setbeamercolor{banner}{fg=bandtext,bg=band}
\setbeamercolor{block body}{fg=bodytext,bg=bodytint}
\setbeamercolor{logostrip}{fg=bandtext,bg=white}

% Semantic emphasis — a colour may mean exactly one thing. Content uses
% these; \hlm marks MEASURED values, \hlc COMPUTED/predicted values,
% \hl colourless emphasis. (\boldmath is a switch: keep it inside the
% argument's group — see docs/conventions/poster-typography.md.)
\newcommand{\hlm}[1]{\textcolor{measured}{\boldmath\textbf{#1}}}
\newcommand{\hlc}[1]{\textcolor{computed}{\boldmath\textbf{#1}}}
\newcommand{\hl}[1]{{\boldmath\textbf{#1}}}
% Citation glue: never wrap away from the cited word (\unskip eats the
% call-site space, \nobreak\ replaces it unbreakably).
\newcommand{\src}[1]{\unskip\nobreak\ {\footnotesize\textcolor{citation}{[#1]}}}
\setbeamerfont{block title}{size=\Large,series=\bfseries}
\setbeamertemplate{block begin}{%
  \vskip0.3ex
  \begin{beamercolorbox}[leftskip=1.2em,rightskip=1.2em,vmode]{block title}%
    % A cap and descender give every title band the same glyph-based height.
    \vskip1.0ex\usebeamerfont{block title}%
    \hyphenpenalty=10000\exhyphenpenalty=10000
    \vphantom{Ay}\insertblocktitle\vskip0.45ex
  \end{beamercolorbox}%
  \begin{beamercolorbox}[leftskip=1.2em,rightskip=1.2em,vmode]{block body}%
    \vskip0.6ex
}
\setbeamertemplate{block end}{\vskip0.8ex\end{beamercolorbox}\vskip0.8ex}

\newlength{\colheight}\setlength{\colheight}{2650pt}

% Logo strip — \PosterStrip stays empty on the bare scaffold; export
% renews it to a \PosterStripBox call (institution logos, then sponsor
% logos, then \hfill + the sponsor line), which renders only then.
\newcommand{\PosterStripBox}[1]{%
  \vspace*{-\topsep}\begin{center}\begin{minipage}{0.95\textwidth}%
  \begin{beamercolorbox}[wd=\linewidth,leftskip=1.2em,rightskip=1.2em,vmode]{logostrip}
    \vskip0.6ex #1\vskip0.6ex
  \end{beamercolorbox}%
  \end{minipage}\end{center}\vspace*{-\topsep}\vskip0.8ex}
